cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(vimar_ota_recovery_migration)
if(NOT DEFINED GENERATED_DIR)
 message(FATAL_ERROR "GENERATED_DIR must contain validated migration bundle headers")
endif()
if(CONFIG_MIGRATION_COMMON AND NOT CONFIG_VIMAR_LOADER_RUNTIME_V2 AND NOT EXISTS "${GENERATED_DIR}/common_platform.h")
 message(FATAL_ERROR "Common migration requires frozen common_platform.h authorization")
endif()
target_sources(app PRIVATE src/main.c src/migration_copy.c ../ota/ota_image.c)
target_include_directories(app PRIVATE src ../ota ${GENERATED_DIR})
target_compile_definitions(app PRIVATE OTA_IMAGE_AUTH_ONLY)
if(CONFIG_MIGRATION_STAGE_PROBE)
 target_sources(app PRIVATE ../ota/ota_storage.c ../ota/ota_transaction.c ../ota/ota_format.c)
endif()
if(CONFIG_MIGRATION_LEGACY_ZIGBEE_ENTRY)
 target_sources(app PRIVATE src/legacy_bridge_policy.c src/legacy_bridge_flow.c
   src/legacy_bridge_entry_zephyr.c ../ota/ota_transaction.c ../src/boot_trace.c)
endif()

if(CONFIG_VIMAR_OTA_SPLIT_STAGING)
 target_sources(app PRIVATE src/migration_policy.c ../src/vimar_power_protection.c ../src/candidate_power_protection.c ../research/src/research_handoff.c)
 target_include_directories(app PRIVATE ../src ../research/src)
endif()

include(${CMAKE_CURRENT_SOURCE_DIR}/../loader_contract/runtime.cmake)
