cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(vimar_common_stock_restore)
if(NOT CONFIG_VIMAR_COMMON_RESTORE_EXPERIMENTAL)
  message(FATAL_ERROR "Common RESTORE is isolated/default OFF; explicit probe configuration required")
endif()
if(NOT DEFINED VIMAR_LOADER_CONTRACT_BINDINGS_HEADER)
  message(FATAL_ERROR "Explicit finite typed loader-family bindings are required")
endif()
target_sources(app PRIVATE main.c platform_zephyr.c common_restore.c protocol.c radio_identity.c monitor.c finalizer_core.c link.c
  ../loader_contract/loader_contract.c ../loader_contract/loader_restore_complete.c ../src/candidate_power_protection.c)
target_include_directories(app PRIVATE . ../loader_contract ../src ../common_restore_gate ../ota)
target_compile_definitions(app PRIVATE VIMAR_LOADER_CONTRACT_BINDINGS_HEADER="${VIMAR_LOADER_CONTRACT_BINDINGS_HEADER}")
zephyr_code_relocate(FILES ${CMAKE_CURRENT_SOURCE_DIR}/../src/candidate_power_protection.c LOCATION RAM_TEXT)
if(CONFIG_VIMAR_COMMON_RESTORE_BOOTSTRAP)
  if(NOT DEFINED VIMAR_COMMON_RESTORE_BLOB_HEADER)
    message(FATAL_ERROR "Explicit immutable RESTORE blob required for stock bootstrap")
  endif()
  target_sources(app PRIVATE bootstrap.c ../common_restore_gate/common_restore_policy.c ../ota/ota_image.c)
  target_compile_definitions(app PRIVATE OTA_IMAGE_AUTH_ONLY VIMAR_COMMON_RESTORE_BLOB_HEADER="${VIMAR_COMMON_RESTORE_BLOB_HEADER}")
endif()
